cmake_minimum_required(VERSION 3.15)
project(SPINDEX C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -Wall -Wextra -g -fPIC")

set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${PROJECT_BINARY_DIR}/lib)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${PROJECT_BINARY_DIR}/lib)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${PROJECT_BINARY_DIR}/bin)

include_directories(include core)

find_package(Threads REQUIRED) # Standard CMake module for pthreads, generally not Conan-managed unless explicit

# --- Dependency Management using Conan ---
# This project is configured to use Conan for managing external C/C++ dependencies.
# Ensure that Conan is installed and that 'conan install' has been run to generate
# the 'conan_toolchain.cmake' and corresponding Conan-generated find modules.
#
# CMake will use the <PackageName>Config.cmake files generated by Conan's CMakeDeps,
# thanks to the CMAKE_TOOLCHAIN_FILE from Conan.
# Using CONFIG mode tells CMake to look specifically for these Config.cmake files.
message(STATUS "--- Looking for Conan-provided dependencies ---")

# REQUIRED dependencies (project will fail to configure if not found via Conan)
find_package(libcurl CONFIG REQUIRED)
find_package(Jansson CONFIG REQUIRED)
find_package(GEOS CONFIG REQUIRED)
find_package(OpenVINO CONFIG REQUIRED) # Assumes OpenVINO is available via a Conan recipe
find_package(libzip CONFIG REQUIRED)
find_package(ZLIB CONFIG REQUIRED)
find_package(Zstd CONFIG REQUIRED)

message(STATUS "--- Conan-provided dependencies search complete ---")
# The actual imported target names (e.g., jansson::jansson, GEOS::geos)
# will be provided by Conan's generated CMake files. These targets
# should be used in target_link_libraries() calls in other CMakeLists.txt files.
# Common convention for Conan targets is <PackageName>::<PackageName>.

# --- End Dependency Management ---

message(STATUS "Adding subdirectories in specific order for dependency resolution.")
add_subdirectory(plugins/aegis_modules) # Defines spindex_aegis (STATIC LIB)
add_subdirectory(plugins)              # Defines all .so plugins and the 'stage_all_plugins_for_core_tests' custom target
add_subdirectory(core)                 # Defines spindex_core (STATIC LIB), spindex_app (EXE), and core test executables
add_subdirectory(tests)                # Defines test_openvino_classifier_hw (EXE) and other standalone tests

enable_testing()
