cmake_minimum_required(VERSION 3.15)
project(SPINDEX VERSION 1.0 LANGUAGES C)

# --- Global Settings ---
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

set(CMAKE_POSITION_INDEPENDENT_CODE ON)

# Default build type if not specified by user (e.g., -DCMAKE_BUILD_TYPE=Release)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Debug)
endif()

# Common compile flags
set(CMAKE_C_FLAGS_INIT "") # Initialize
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS_INIT} -Wall -Wextra")
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
  set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -g")
endif()
# -fPIC is handled by CMAKE_POSITION_INDEPENDENT_CODE ON

# Output directories
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib/spindex/plugins) # For plugins
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)

# --- Include Directories ---
# Add top-level include directory globally.
# Target-specific include directories are preferred but this is a starting point.
include_directories(
    "${CMAKE_CURRENT_SOURCE_DIR}/include"
    "${CMAKE_CURRENT_SOURCE_DIR}/core" # core might have internal headers
    "${CMAKE_CURRENT_SOURCE_DIR}/plugins/aegis_modules" # AEGIS module headers
)

# --- External Dependencies ---
# When using Conan, these find_package calls will use the <PackageName>Config.cmake
# files generated by Conan's CMakeDeps generator in the build_cmake/generators directory.
# The CMAKE_TOOLCHAIN_FILE (conan_toolchain.cmake) handles setting up paths for CMake.

# Threads (Still use CMake's built-in FindThreads as it's fundamental)
find_package(Threads REQUIRED)

# Try Conan first, fall back to pkg-config for system packages
find_package(CURL CONFIG QUIET)
if(NOT CURL_FOUND)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(CURL REQUIRED libcurl)
    # Create imported target for compatibility
    add_library(CURL::libcurl INTERFACE IMPORTED)
    set_target_properties(CURL::libcurl PROPERTIES
        INTERFACE_INCLUDE_DIRECTORIES "${CURL_INCLUDE_DIRS}"
        INTERFACE_LINK_LIBRARIES "${CURL_LIBRARIES}"
    )
endif()

find_package(Jansson CONFIG QUIET)
if(NOT Jansson_FOUND)
    pkg_check_modules(JANSSON REQUIRED jansson)
    add_library(jansson::jansson INTERFACE IMPORTED)
    set_target_properties(jansson::jansson PROPERTIES
        INTERFACE_INCLUDE_DIRECTORIES "${JANSSON_INCLUDE_DIRS}"
        INTERFACE_LINK_LIBRARIES "${JANSSON_LIBRARIES}"
    )
endif()

find_package(GEOS CONFIG QUIET)
if(NOT GEOS_FOUND)
    pkg_check_modules(GEOS REQUIRED geos)
    add_library(GEOS::geos_c INTERFACE IMPORTED)
    set_target_properties(GEOS::geos_c PROPERTIES
        INTERFACE_INCLUDE_DIRECTORIES "${GEOS_INCLUDE_DIRS}"
        INTERFACE_LINK_LIBRARIES "${GEOS_LIBRARIES}"
    )
endif()

# OpenVINO (Handled by Conan)
option(USE_OPENVINO "Enable OpenVINO support" ON)
if(USE_OPENVINO)
    # The OpenVINO package from Conan should provide an OpenVINOConfig.cmake
    # or openvino-config.cmake. The name "OpenVINO" here must match what Conan's
    # CMakeDeps generates (check build_cmake/generators/*Config.cmake).
    # It might be case-sensitive or hyphenated depending on the recipe.
    # Assuming "OpenVINO" for now.
    find_package(OpenVINO CONFIG) # Try without REQUIRED first to handle gracefully
    if(OpenVINO_FOUND)
        message(STATUS "Found OpenVINO via Conan.")
        # OPENVINO_FOUND is set by find_package.
        # OpenVINO::OpenVINO (or similar target) should be available for linking.
        # Set a global variable for convenience in subdirectories if needed, though linking targets is preferred.
        # set(OPENVINO_CONAN_FOUND TRUE CACHE INTERNAL "OpenVINO found via Conan")
    else()
        message(WARNING "OpenVINO Conan package was requested (USE_OPENVINO=ON) but not found by find_package(OpenVINO CONFIG). Ensure the OpenVINO Conan package is correctly installed and provides CMake configuration files. OpenVINO dependent components may fail to build.")
        set(OPENVINO_FOUND FALSE) # Explicitly set if find_package fails
    endif()
else()
    message(STATUS "OpenVINO support is disabled (USE_OPENVINO=OFF).")
    set(OPENVINO_FOUND FALSE)
endif()
