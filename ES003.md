## Overview

This guide provides comprehensive instructions for implementing a high-performance hybrid search system combining FAISS (vector search) and Tantivy (text search) for the SPINDEX project. The implementation leverages AVX-512 instructions for optimal performance on modern Intel processors.

### Key Features
- **Dual-index architecture**: Separate optimized indices for text and vector search
- **AVX-512 optimization**: Hardware-accelerated similarity computations
- **Unified API**: Single interface for hybrid queries
- **Synchronization**: Atomic updates across both indices
- **Migration support**: Seamless transition from Elasticsearch

### Performance Targets
- Query latency: <500μs (p95)
- Index throughput: >50k documents/second
- Memory efficiency: <8GB for 10M documents
- Search accuracy: >0.85 NDCG@10

```rust
use std::ffi::{CStr, CString};
use std::os::raw::{c_char, c_void};
use std::ptr;
use tantivy::collector::TopDocs;
use tantivy::query::QueryParser;
use tantivy::schema::*;
use tantivy::{doc, Index, IndexReader, IndexWriter, TantivyError};

#[repr(C)]
pub struct TantivyIndex {
    index: Index,
    reader: IndexReader,
    writer: Option<IndexWriter>,
    schema: Schema,
    content_field: Field,
    id_field: Field,
}

#[no_mangle]
pub extern "C" fn tantivy_create_index(
    path: *const c_char,
) -> *mut TantivyIndex {
    let path_str = unsafe {
        assert!(!path.is_null());
        CStr::from_ptr(path).to_string_lossy().into_owned()
    };

    // Build schema
    let mut schema_builder = Schema::builder();
    let content_field = schema_builder.add_text_field(
        "content",
        TextOptions::default()
            .set_indexing_options(
                TextFieldIndexing::default()
                    .set_tokenizer("default")
                    .set_index_option(IndexRecordOption::WithFreqsAndPositions),
            )
            .set_stored(),
    );
    let id_field = schema_builder.add_u64_field("id", INDEXED | STORED);
    let file_type_field = schema_builder.add_text_field("file_type", STRING | STORED);
    
    let schema = schema_builder.build();

    // Create or open index
    let index = match Index::open_in_dir(&path_str) {
        Ok(index) => index,
        Err(_) => {
            std::fs::create_dir_all(&path_str).ok();
            Index::create_in_dir(&path_str, schema.clone()).unwrap()
        }
    };

    let reader = index.reader().unwrap();
    let writer = index.writer(100_000_000).ok(); // 100MB buffer

    Box::into_raw(Box::new(TantivyIndex {
        index,
        reader,
        writer,
        schema,
        content_field,
        id_field,
    }))
}

#[no_mangle]
pub extern "C" fn tantivy_index_document(
    index_ptr: *mut TantivyIndex,
    doc_id: u64,
    content: *const c_char,
    file_type: *const c_char,
) -> i32 {
    let index = unsafe {
        assert!(!index_ptr.is_null());
        &mut *index_ptr
    };

    let content_str = unsafe {
        assert!(!content.is_null());
        CStr::from_ptr(content).to_str().unwrap()
    };

    let file_type_str = unsafe {
        assert!(!file_type.is_null());
        CStr::from_ptr(file_type).to_str().unwrap()
    };

    if let Some(ref mut writer) = index.writer {
        let mut doc = Document::default();
        doc.add_u64(index.id_field, doc_id);
        doc.add_text(index.content_field, content_str);
        doc.add_text(
            index.schema.get_field("file_type").unwrap(), 
            file_type_str
        );
        
        writer.add_document(doc).unwrap();
        0 // Success
    } else {
        -1 // No writer available
    }
}

#[no_mangle]
pub extern "C" fn tantivy_search(
    index_ptr: *mut TantivyIndex,
    query_str: *const c_char,
    top_k: usize,
    results: *mut u64,
    scores: *mut f32,
) -> i32 {
    let index = unsafe {
        assert!(!index_ptr.is_null());
        &*index_ptr
    };

    let query_text = unsafe {
        assert!(!query_str.is_null());
        CStr::from_ptr(query_str).to_str().unwrap()
    };

    let searcher = index.reader.searcher();
    let query_parser = QueryParser::for_index(
        &index.index, 
        vec![index.content_field]
    );
    
    match query_parser.parse_query(query_text) {
        Ok(query) => {
            let top_docs = searcher
                .search(&query, &TopDocs::with_limit(top_k))
                .unwrap();

            for (i, (score, doc_address)) in top_docs.iter().enumerate() {
                let doc = searcher.doc(*doc_address).unwrap();
                let doc_id = doc.get_first(index.id_field)
                    .unwrap()
                    .as_u64()
                    .unwrap();
                
                unsafe {
                    *results.add(i) = doc_id;
                    *scores.add(i) = *score;
                }
            }
            
            top_docs.len() as i32
        }
        Err(_) => -1,
    }
}

#[no_mangle]
pub extern "C" fn tantivy_commit(index_ptr: *mut TantivyIndex) -> i32 {
    let index = unsafe {
        assert!(!index_ptr.is_null());
        &mut *index_ptr
    };

    if let Some(ref mut writer) = index.writer {
        match writer.commit() {
            Ok(_) => {
                // Reload reader to see new documents
                index.reader.reload().ok();
                0
            }
            Err(_) => -1,
        }
    } else {
        -1
    }
}

#[no_mangle]
pub extern "C" fn tantivy_destroy(index_ptr: *mut TantivyIndex) {
    if !index_ptr.is_null() {
        unsafe {
            let _ = Box::from_raw(index_ptr);
        }
    }
}
```

### Step 4: FAISS Integration

```c
// src/faiss_wrapper.c
#include <faiss/IndexFlat.h>
#include <faiss/IndexIVFPQ.h>
#include <faiss/index_io.h>
#include <faiss/impl/AuxIndexStructures.h>
#include <immintrin.h>
#include <string.h>
#include <stdio.h>

typedef struct {
    faiss::Index* index;
    size_t dimension;
    bool use_avx512;
    char* index_path;
} faiss_index_t;

// AVX-512 optimized distance computation
static float avx512_l2_distance(const float* a, const float* b, size_t d) {
    __m512 sum = _mm512_setzero_ps();
    
    size_t i;
    for (i = 0; i + 16 <= d; i += 16) {
        __m512 va = _mm512_loadu_ps(&a[i]);
        __m512 vb = _mm512_loadu_ps(&b[i]);
        __m512 diff = _mm512_sub_ps(va, vb);
        sum = _mm512_fmadd_ps(diff, diff, sum);
    }
    
    // Handle remaining elements
    float tail_sum = 0.0f;
    for (; i < d; i++) {
        float diff = a[i] - b[i];
        tail_sum += diff * diff;
    }
    
    // Horizontal sum of AVX-512 register
    float avx_sum = _mm512_reduce_add_ps(sum);
    return avx_sum + tail_sum;
}

extern "C" {

faiss_index_t* faiss_create_index(const char* path, size_t dimension, 
                                  const char* index_type) {
    faiss_index_t* wrapper = (faiss_index_t*)malloc(sizeof(faiss_index_t));
    wrapper->dimension = dimension;
    wrapper->use_avx512 = __builtin_cpu_supports("avx512f");
    wrapper->index_path = strdup(path);
    
    // Create appropriate index type
    if (strcmp(index_type, "Flat") == 0) {
        wrapper->index = new faiss::IndexFlatL2(dimension);
    } else if (strcmp(index_type, "IVF") == 0) {
        // Create IVF index with PQ compression
        int nlist = 1024;  // Number of clusters
        int m = 64;        // Number of subquantizers
        wrapper->index = new faiss::IndexIVFPQ(
            new faiss::IndexFlatL2(dimension),
            dimension, nlist, m, 8
        );
    } else {
        // Default to Flat index
        wrapper->index = new faiss::IndexFlatL2(dimension);
    }
    
    // Try to load existing index
    try {
        *wrapper->index = *faiss::read_index(path);
    } catch (...) {
        // New index, will be saved later
    }
    
    return wrapper;
}

int faiss_add_vectors(faiss_index_t* index, size_t n, 
                     const float* vectors, const int64_t* ids) {
    if (!index || !index->index) return -1;
    
    try {
        if (ids) {
            // Use ID mapper for direct ID support
            faiss::IndexIDMap* id_index = 
                dynamic_cast<faiss::IndexIDMap*>(index->index);
            if (!id_index) {
                // Wrap existing index with ID mapper
                id_index = new faiss::IndexIDMap(index->index);
                index->index = id_index;
            }
            id_index->add_with_ids(n, vectors, ids);
        } else {
            index->index->add(n, vectors);
        }
        return 0;
    } catch (const faiss::FaissException& e) {
        fprintf(stderr, "FAISS error: %s\n", e.what());
        return -1;
    }
}

int faiss_search(faiss_index_t* index, size_t n_queries,
                const float* queries, size_t k,
                int64_t* labels, float* distances) {
    if (!index || !index->index) return -1;
    
    try {
        if (index->use_avx512) {
            // Custom AVX-512 search for Flat index
            faiss::IndexFlat* flat_index = 
                dynamic_cast<faiss::IndexFlat*>(index->index);
            if (flat_index) {
                // Implement custom AVX-512 search
                #pragma omp parallel for
                for (size_t q = 0; q < n_queries; q++) {
                    const float* query = queries + q * index->dimension;
                    
                    // Compute distances to all vectors
                    std::vector<std::pair<float, int64_t>> dists;
                    for (int64_t i = 0; i < flat_index->ntotal; i++) {
                        const float* vec = flat_index->xb.data() + 
                                          i * index->dimension;
                        float dist = avx512_l2_distance(query, vec, 
                                                       index->dimension);
                        dists.push_back({dist, i});
                    }
                    
                    // Find top-k
                    std::partial_sort(dists.begin(), 
                                     dists.begin() + k,
                                     dists.end());
                    
                    // Copy results
                    for (size_t i = 0; i < k; i++) {
                        distances[q * k + i] = dists[i].first;
                        labels[q * k + i] = dists[i].second;
                    }
                }
                return 0;
            }
        }
        
        // Fall back to standard FAISS search
        index->index->search(n_queries, queries, k, distances, labels);
        return 0;
    } catch (const faiss::FaissException& e) {
        fprintf(stderr, "FAISS search error: %s\n", e.what());
        return -1;
    }
}

int faiss_save_index(faiss_index_t* index) {
    if (!index || !index->index) return -1;
    
    try {
        faiss::write_index(index->index, index->index_path);
        return 0;
    } catch (const faiss::FaissException& e) {
        fprintf(stderr, "FAISS save error: %s\n", e.what());
        return -1;
    }
}

void faiss_destroy_index(faiss_index_t* index) {
    if (index) {
        delete index->index;
        free(index->index_path);
        free(index);
    }
}

} // extern "C"
```

```c
// Internal structures
typedef struct {
    uint64_t doc_id;
    float text_score;
    float vector_score;
    float combined_score;
} score_entry_t;

// Utility functions
static double get_time_ms() {
    struct timeval tv;
    gettimeofday(&tv, NULL);
    return tv.tv_sec * 1000.0 + tv.tv_usec / 1000.0;
}

// Score normalization (0-1 range)
static void normalize_scores(float* scores, size_t count) {
    if (count == 0) return;
    
    float min_score = scores[0], max_score = scores[0];
    for (size_t i = 1; i < count; i++) {
        if (scores[i] < min_score) min_score = scores[i];
        if (scores[i] > max_score) max_score = scores[i];
    }
    
    float range = max_score - min_score;
    if (range > 0) {
        for (size_t i = 0; i < count; i++) {
            scores[i] = (scores[i] - min_score) / range;
        }
    }
}

// Reciprocal Rank Fusion
static void reciprocal_rank_fusion(score_entry_t* entries, size_t count,
                                  float text_weight, float vector_weight) {
    const float k = 60.0f; // RRF constant
    
    for (size_t i = 0; i < count; i++) {
        float text_rrf = 1.0f / (k + i + 1);
        float vector_rrf = 1.0f / (k + i + 1);
        
        entries[i].combined_score = 
            text_weight * text_rrf + vector_weight * vector_rrf;
    }
}

// Main implementation
hybrid_search_engine_t* hybrid_search_create(const char* config_path) {
    hybrid_search_engine_t* engine = 
        (hybrid_search_engine_t*)calloc(1, sizeof(hybrid_search_engine_t));
    
    // Parse configuration (simplified for this example)
    engine->index_path = strdup("/data/hybrid_index");
    engine->vector_dim = 768; // DistilBERT dimension
    engine->use_avx512 = true;
    
    // Initialize indices
    char text_path[256], vector_path[256];
    snprintf(text_path, sizeof(text_path), "%s/tantivy", engine->index_path);
    snprintf(vector_path, sizeof(vector_path), "%s/faiss", engine->index_path);
    
    engine->text_index = tantivy_create_index(text_path);
    engine->vector_index = faiss_create_index(vector_path, 
                                             engine->vector_dim, 
                                             "IVF");
    
    // Initialize synchronization
    pthread_rwlock_init(&engine->sync_lock, NULL);
    
    return engine;
}

int hybrid_index_document(hybrid_search_engine_t* engine,
                         const document_t* doc) {
    if (!engine || !doc) return -1;
    
    pthread_rwlock_wrlock(&engine->sync_lock);
    
    // Index text
    int text_result = tantivy_index_document(
        engine->text_index,
        doc->id,
        doc->content,
        doc->file_type
    );
    
    if (text_result != 0) {
        pthread_rwlock_unlock(&engine->sync_lock);
        return -1;
    }
    
    // Index vector
    int64_t id = (int64_t)doc->id;
    int vector_result = faiss_add_vectors(
        engine->vector_index,
        1,
        doc->embedding,
        &id
    );
    
    if (vector_result != 0) {
        // TODO: Rollback text index
        pthread_rwlock_unlock(&engine->sync_lock);
        return -1;
    }
    
    engine->total_docs++;
    pthread_rwlock_unlock(&engine->sync_lock);
    
    return 0;
}

int hybrid_index_batch(hybrid_search_engine_t* engine,
                      const document_t* docs,
                      size_t count) {
    if (!engine || !docs || count == 0) return -1;
    
    pthread_rwlock_wrlock(&engine->sync_lock);
    
    // Batch index text
    for (size_t i = 0; i < count; i++) {
        int result = tantivy_index_document(
            engine->text_index,
            docs[i].id,
            docs[i].content,
            docs[i].file_type
        );
        if (result != 0) {
            pthread_rwlock_unlock(&engine->sync_lock);
            return -1;
        }
    }
    
    // Commit text changes
    tantivy_commit(engine->text_index);
    
    // Prepare vectors for batch indexing
    float* all_vectors = (float*)aligned_alloc(
        64, count * engine->vector_dim * sizeof(float)
    );
    int64_t* all_ids = (int64_t*)malloc(count * sizeof(int64_t));
    
    for (size_t i = 0; i < count; i++) {
        memcpy(all_vectors + i * engine->vector_dim,
               docs[i].embedding,
               engine->vector_dim * sizeof(float));
        all_ids[i] = (int64_t)docs[i].id;
    }
    
    // Batch index vectors
    int vector_result = faiss_add_vectors(
        engine->vector_index,
        count,
        all_vectors,
        all_ids
    );
    
    free(all_vectors);
    free(all_ids);
    
    if (vector_result != 0) {
        pthread_rwlock_unlock(&engine->sync_lock);
        return -1;
    }
    
    // Save vector index
    faiss_save_index(engine->vector_index);
    
    engine->total_docs += count;
    pthread_rwlock_unlock(&engine->sync_lock);
    
    return 0;
}

search_results_t* hybrid_search(hybrid_search_engine_t* engine,
                               const query_t* query) {
    if (!engine || !query) return NULL;
    
    double start_time = get_time_ms();
    
    pthread_rwlock_rdlock(&engine->sync_lock);
    
    // Allocate result buffers
    size_t buffer_size = query->top_k * 2; // Oversample
    uint64_t* text_ids = (uint64_t*)malloc(buffer_size * sizeof(uint64_t));
    float* text_scores = (float*)malloc(buffer_size * sizeof(float));
    int64_t* vector_ids = (int64_t*)malloc(buffer_size * sizeof(int64_t));
    float* vector_scores = (float*)malloc(buffer_size * sizeof(float));
    
    // Parallel search execution
    int text_count = 0, vector_count = 0;
    
    #pragma omp parallel sections
    {
        #pragma omp section
        {
            // Text search
            if (query->text_query && strlen(query->text_query) > 0) {
                text_count = tantivy_search(
                    engine->text_index,
                    query->text_query,
                    buffer_size,
                    text_ids,
                    text_scores
                );
                if (text_count > 0) {
                    normalize_scores(text_scores, text_count);
                }
            }
        }
        
        #pragma omp section
        {
            // Vector search
            if (query->vector_query) {
                int result = faiss_search(
                    engine->vector_index,
                    1,
                    query->vector_query,
                    buffer_size,
                    vector_ids,
                    vector_scores
                );
                if (result == 0) {
                    vector_count = buffer_size;
                    // Convert distances to similarities
                    for (size_t i = 0; i < vector_count; i++) {
                        vector_scores[i] = 1.0f / (1.0f + vector_scores[i]);
                    }
                    normalize_scores(vector_scores, vector_count);
                }
            }
        }
    }
    
    pthread_rwlock_unlock(&engine->sync_lock);
    
    // Merge results
    score_entry_t* entries = (score_entry_t*)calloc(
        text_count + vector_count, sizeof(score_entry_t)
    );
    size_t entry_count = 0;
    
    // Create score map
    for (int i = 0; i < text_count; i++) {
        entries[entry_count].doc_id = text_ids[i];
        entries[entry_count].text_score = text_scores[i];
        entries[entry_count].vector_score = 0.0f;
        entry_count++;
    }
    
    // Merge vector scores
    for (int i = 0; i < vector_count; i++) {
        bool found = false;
        for (size_t j = 0; j < entry_count; j++) {
            if (entries[j].doc_id == (uint64_t)vector_ids[i]) {
                entries[j].vector_score = vector_scores[i];
                found = true;
                break;
            }
        }
        if (!found && entry_count < text_count + vector_count) {
            entries[entry_count].doc_id = (uint64_t)vector_ids[i];
            entries[entry_count].text_score = 0.0f;
            entries[entry_count].vector_score = vector_scores[i];
            entry_count++;
        }
    }
    
    // Apply fusion
    reciprocal_rank_fusion(entries, entry_count,
                          query->text_weight,
                          query->vector_weight);
    
    // Sort by combined score
    qsort(entries, entry_count, sizeof(score_entry_t),
          [](const void* a, const void* b) {
              score_entry_t* ea = (score_entry_t*)a;
              score_entry_t* eb = (score_entry_t*)b;
              return (eb->combined_score > ea->combined_score) ? 1 : -1;
          });
    
    // Build final results
    search_results_t* results = 
        (search_results_t*)malloc(sizeof(search_results_t));
    results->count = (entry_count < query->top_k) ? entry_count : query->top_k;
    results->results = (result_t*)malloc(results->count * sizeof(result_t));
    
    for (size_t i = 0; i < results->count; i++) {
        results->results[i].doc_id = entries[i].doc_id;
        results->results[i].score = entries[i].combined_score;
        results->results[i].snippet = NULL; // TODO: Generate snippets
    }
    
    results->search_time_ms = get_time_ms() - start_time;
    
    // Update statistics
    engine->total_queries++;
    engine->avg_query_time_ms = 
        (engine->avg_query_time_ms * (engine->total_queries - 1) + 
         results->search_time_ms) / engine->total_queries;
    
    // Cleanup
    free(text_ids);
    free(text_scores);
    free(vector_ids);
    free(vector_scores);
    free(entries);
    
    return results;
}

void search_results_free(search_results_t* results) {
    if (results) {
        if (results->results) {
            for (size_t i = 0; i < results->count; i++) {
                free(results->results[i].snippet);
            }
            free(results->results);
        }
        free(results);
    }
}

void hybrid_search_destroy(hybrid_search_engine_t* engine) {
    if (engine) {
        pthread_rwlock_destroy(&engine->sync_lock);
        tantivy_destroy(engine->text_index);
        faiss_destroy_index(engine->vector_index);
        free(engine->index_path);
        free(engine);
    }
}
```

```c
        (hybrid_search_plugin_t*)malloc(sizeof(hybrid_search_plugin_t));

    hybrid_search_plugin_t* plugin = (hybrid_search_plugin_t*)plugin_ptr;

    return hybrid_search(plugin->engine, query);
```

```c
typedef struct {
    void* base;
    size_t size;
    size_t used;
    pthread_mutex_t lock;
} memory_pool_t;

memory_pool_t* create_memory_pool(size_t size) {
    memory_pool_t* pool = malloc(sizeof(memory_pool_t));
    pool->base = aligned_alloc(64, size); // Cache line aligned
    pool->size = size;
    pool->used = 0;
    pthread_mutex_init(&pool->lock, NULL);
    
    // Pre-fault pages
    memset(pool->base, 0, size);
    
    // Advise kernel for huge pages
    madvise(pool->base, size, MADV_HUGEPAGE);
    
    return pool;
}
```
